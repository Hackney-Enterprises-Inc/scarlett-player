<div
  data-scarlett-player
  data-src="https://example.com/film.m3u8"
  data-brand-color="#e50914"
></div>
<script src="https://assets.thestreamplatform.com/scarlett-player/latest/embed.video.umd.cjs"></script>
// npm install @scarlett-player/core @scarlett-player/hls @scarlett-player/ui
import { createPlayer } from '@scarlett-player/core';
import { createHLSPlugin } from '@scarlett-player/hls';
import { uiPlugin } from '@scarlett-player/ui';

const player = await createPlayer({
  container: '#player',
  src: 'https://example.com/film.m3u8',
  plugins: [
    createHLSPlugin(),
    uiPlugin({ theme: { accentColor: '#e50914' } }),
  ],
});

// Later, when the page tears down:
player.destroy();
<script setup lang="ts">
// npm install @scarlett-player/vue @scarlett-player/hls @scarlett-player/ui
import { ScarlettPlayerComponent as ScarlettPlayer } from '@scarlett-player/vue';
import { createHLSPlugin } from '@scarlett-player/hls';
import { uiPlugin } from '@scarlett-player/ui';

const plugins = [
  createHLSPlugin(),
  uiPlugin({ theme: { accentColor: '#e50914' } }),
];
</script>

<template>
  <ScarlettPlayer src="https://example.com/film.m3u8" :plugins="plugins" />
</template>
